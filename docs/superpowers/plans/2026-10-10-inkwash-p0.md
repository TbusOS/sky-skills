- [ ] **Step 1: 写脚本**(以下代码 2026-10-10 在临时目录实跑过:自测 13 项通过,两处故意改坏都失败)

`$W/skills/design-skill-method/scripts/check_registration.sh`:

```bash
#!/usr/bin/env bash
# check_registration.sh — 一个 skill 在仓库里该登记的地方,是不是都登记了。
#
# 为什么要有它:加一个设计 skill 要同步改十几个文件(检查脚本的 skill 列表、
# 名册、首页、README、critic 配置……)。这份清单以前只写在设计书里,
# 2026-10-10 实测 graphite 漏了 15 处、relief 漏了 9 处,没有任何东西报错。
#
# 用法:
#   check_registration.sh --repo <仓库根> --skill <目录名> [--kind style|tool]
#                         [--registry <tsv>] [--exempt <tsv>]
#   check_registration.sh --selftest
#
#   --skill    skills/ 下的目录名,如 graphite-design;{short} 是去掉 -design 的部分
#   --kind     style = 设计风格 skill(查全部注册点,默认);tool = 其他 skill(只查 kind=any 的)
#   --registry 注册点清单,默认 ../params/sky-skills/registry.tsv
#   --exempt   豁免清单,默认 ../params/sky-skills/exemptions.tsv
#
# 注册点清单(TSV,# 开头是注释):
#   id  kind(any|style)  文件(相对仓库根)  扩展正则({name} {short} 会被替换)  至少几行  说明
# 豁免清单(TSV):skill目录名  注册点id  理由(不能为空)
#
# 输出:每个注册点一行 OK / MISS / EXEMPT / N/A / NOFILE,最后一行汇总。
# 退出码:0 全部 OK 或豁免 · 1 有 MISS 或 NOFILE · 2 参数错 · 3 豁免没写理由
set -u

HERE="$(cd "$(dirname "$0")" && pwd)"

usage() { sed -n '9,16p' "$0" | sed 's/^# \{0,1\}//'; }

run_check() {
  local repo="" skill="" kind="style"
  local registry="$HERE/../params/sky-skills/registry.tsv"
  local exempt="$HERE/../params/sky-skills/exemptions.tsv"
  while [ $# -gt 0 ]; do
    case "$1" in
      --repo) repo="${2:-}"; shift 2 ;;
      --skill) skill="${2:-}"; shift 2 ;;
      --kind) kind="${2:-}"; shift 2 ;;
      --registry) registry="${2:-}"; shift 2 ;;
      --exempt) exempt="${2:-}"; shift 2 ;;
      -h|--help) usage; return 0 ;;
      *) echo "未知参数: $1" >&2; usage >&2; return 2 ;;
    esac
  done
  [ -n "$repo" ] && [ -d "$repo" ] || { echo "--repo 不是目录: '$repo'" >&2; return 2; }
  echo "$skill" | grep -Eq '^[a-z][a-z0-9-]*$' || { echo "--skill 只能是小写字母、数字和连字符: '$skill'" >&2; return 2; }
  [ "$kind" = style ] || [ "$kind" = tool ] || { echo "--kind 只能是 style 或 tool: '$kind'" >&2; return 2; }
  [ -f "$registry" ] || { echo "找不到注册点清单: $registry" >&2; return 2; }
  local short="${skill%-design}"

  # 豁免:理由为空直接报错 —— 不写理由的豁免等于悄悄跳过。
  local exempt_ids="" bad_exempt=0
  if [ -f "$exempt" ]; then
    while IFS=$'\t' read -r e_skill e_id e_reason; do
      case "$e_skill" in ''|'#'*) continue ;; esac
      [ "$e_skill" = "$skill" ] || continue
      if [ -z "${e_reason// /}" ]; then
        echo "豁免 $e_skill $e_id 没写理由" >&2; bad_exempt=1
      fi
      exempt_ids="$exempt_ids $e_id"
    done < "$exempt"
  fi
  [ "$bad_exempt" = 0 ] || return 3

  local ok=0 miss=0 ex=0 na=0 nofile=0
  while IFS=$'\t' read -r id pkind file pattern min what; do
    case "$id" in ''|'#'*) continue ;; esac
    if [ "$kind" = tool ] && [ "$pkind" = style ]; then
      printf 'N/A     %-4s %s\n' "$id" "$what"; na=$((na + 1)); continue
    fi
    case " $exempt_ids " in
      *" $id "*)
        local why
        why="$(awk -F'\t' -v s="$skill" -v i="$id" '$1==s && $2==i {print $3}' "$exempt")"
        printf 'EXEMPT  %-4s %s(%s)\n' "$id" "$what" "$why"; ex=$((ex + 1)); continue ;;
    esac
    if [ ! -f "$repo/$file" ]; then
      printf 'NOFILE  %-4s %s\n' "$id" "$file"; nofile=$((nofile + 1)); continue
    fi
    local re="${pattern//\{name\}/$skill}"
    re="${re//\{short\}/$short}"
    local n
    n="$(grep -Ec -- "$re" "$repo/$file")"
    if [ "$n" -ge "${min:-1}" ]; then
      printf 'OK      %-4s %s\n' "$id" "$what"; ok=$((ok + 1))
    else
      printf 'MISS    %-4s %s —— %s 里匹配 %s 行,至少要 %s 行\n' "$id" "$what" "$file" "$n" "${min:-1}"
      miss=$((miss + 1))
    fi
  done < "$registry"

  echo "汇总 $skill: OK $ok · MISS $miss · NOFILE $nofile · EXEMPT $ex · N/A $na"
  [ $((miss + nofile)) -eq 0 ]
}

selftest() {
  local t fail=0
  t="$(mktemp -d)"
  trap 'rm -rf "$t"' RETURN
  mkdir -p "$t/repo" "$t/has space"
  printf 'SKILLS = {\n  "foo": {\n# "qux": { 只在注释里提到,不算登记\n' > "$t/repo/a.py"
  cp "$t/repo/a.py" "$t/has space/a.py"
  printf 'const VALID = [%s];\n' "'foo', 'bar'" > "$t/repo/b.mjs"
  printf '| [foo-design](skills/foo-design/) | x |\n' > "$t/repo/README.md"
  printf 'message passing; foo here\n' > "$t/repo/c.md"
  {
    printf '# 注释行要跳过\n'
    printf 'R01\tstyle\ta.py\t^[[:space:]]*"{short}":[[:space:]]*\\{\t1\ta.py 字典\n'
    printf 'R02\tstyle\tb.mjs\tVALID = \\[.*'"'"'{short}'"'"'\t1\tb.mjs 列表\n'
    printf 'R03\tany\tREADME.md\t^\\| \\[{name}\\]\\(skills/{name}/\\)\t1\tREADME 行\n'
    printf 'R04\tany\tgone.txt\t{short}\t1\t不存在的文件\n'
    printf 'R05\tany\tc.md\t(^|[^a-z]){short}([^a-z]|$)\t1\t整词出现\n'
  } > "$t/reg.tsv"
  : > "$t/ex-empty.tsv"
  printf 'foo-design\tR04\t测试用:这个文件本来就不存在\n' > "$t/ex-ok.tsv"
  printf 'foo-design\tR04\t   \n' > "$t/ex-bad.tsv"

  expect() {  # expect <名字> <期望退出码> <期望输出里要有的串> -- <参数…>
    local name="$1" code="$2" want="$3"; shift 4
    local out rc
    out="$(run_check "$@" 2>&1)"; rc=$?
    if [ "$rc" != "$code" ] || ! printf '%s' "$out" | grep -Fq -- "$want"; then
      echo "FAIL $name: 退出码 $rc(要 $code),输出里要有「${want}」"; printf '%s\n' "$out" | sed 's/^/    /'
      fail=1
    else
      echo "PASS $name"
    fi
  }
  local base=(--repo "$t/repo" --registry "$t/reg.tsv")
  expect "登记全的 skill,文件缺失要报 NOFILE" 1 "NOFILE  R04" -- "${base[@]}" --skill foo-design --exempt "$t/ex-empty.tsv"
  expect "登记全的 skill 四处 OK" 1 "OK 4" -- "${base[@]}" --skill foo-design --exempt "$t/ex-empty.tsv"
  expect "豁免带理由时通过" 0 "EXEMPT  R04" -- "${base[@]}" --skill foo-design --exempt "$t/ex-ok.tsv"
  expect "豁免没写理由要报错" 3 "没写理由" -- "${base[@]}" --skill foo-design --exempt "$t/ex-bad.tsv"
  expect "只在列表里出现的 bar 缺字典和 README" 1 "MISS    R01" -- "${base[@]}" --skill bar-design --exempt "$t/ex-empty.tsv"
  expect "bar 在列表里那处算 OK" 1 "OK      R02" -- "${base[@]}" --skill bar-design --exempt "$t/ex-empty.tsv"
  expect "foobar 不能被 foo 的登记冒充" 1 "MISS    R02" -- "${base[@]}" --skill foobar-design --exempt "$t/ex-empty.tsv"
```

`$SP/spike/brush-proto.js`:

```js
/* 小样用的毛笔原型(一次性,验证完就丢)。
 * 一笔 = 中心线 + 粗细曲线 + 含墨量。湿的部分是填色形状,干的部分是一束笔毛断成的线段(飞白),
 * 整笔外面套一层遮罩,遮罩沿笔路展开,露出下面整笔;落定后晕边层慢慢显出来。
 * 浏览器和 node 都能用:浏览器里是 window.BrushProto,node 里 require。 */
(function (root, factory) {
  if (typeof module === 'object' && module.exports) module.exports = factory();
  else root.BrushProto = factory();
})(this, function () {
  'use strict';

  function rng(seed) {                       // mulberry32:同一个 seed 永远出同一串数
    var a = seed >>> 0;
    return function () {
      a = (a + 0x6D2B79F5) >>> 0;
      var t = a;
      t = Math.imul(t ^ (t >>> 15), t | 1);
      t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
      return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    };
  }

  function noise1d(r, knots) {               // 0..1 上的平滑随机曲线,值在 -1..1
    var v = [];
    for (var i = 0; i <= knots; i++) v.push(r() * 2 - 1);
    return function (t) {
      var x = Math.max(0, Math.min(0.999999, t)) * knots, i = Math.floor(x), f = x - i;
      var s = f * f * (3 - 2 * f);
      return v[i] * (1 - s) + v[i + 1] * s;
    };
  }

  // Catmull-Rom 过所有点,按约 step px 取样;返回 [{x, y, s}],s 是从起点算的长度
  function resample(pts, step) {
    var dense = [];
    for (var i = 0; i < pts.length - 1; i++) {
      var p0 = pts[Math.max(0, i - 1)], p1 = pts[i], p2 = pts[i + 1], p3 = pts[Math.min(pts.length - 1, i + 2)];
      for (var k = 0; k < 24; k++) {
        var t = k / 24, t2 = t * t, t3 = t2 * t;
        dense.push([
          0.5 * (2 * p1[0] + (-p0[0] + p2[0]) * t + (2 * p0[0] - 5 * p1[0] + 4 * p2[0] - p3[0]) * t2 + (-p0[0] + 3 * p1[0] - 3 * p2[0] + p3[0]) * t3),
          0.5 * (2 * p1[1] + (-p0[1] + p2[1]) * t + (2 * p0[1] - 5 * p1[1] + 4 * p2[1] - p3[1]) * t2 + (-p0[1] + 3 * p1[1] - 3 * p2[1] + p3[1]) * t3)
        ]);
      }
    }
    dense.push(pts[pts.length - 1]);
    var out = [{ x: dense[0][0], y: dense[0][1], s: 0 }], acc = 0, last = dense[0];
    for (var j = 1; j < dense.length; j++) {
      var d = Math.hypot(dense[j][0] - last[0], dense[j][1] - last[1]);
      acc += d; last = dense[j];
      if (acc - out[out.length - 1].s >= step || j === dense.length - 1) out.push({ x: dense[j][0], y: dense[j][1], s: acc });
    }
    return out;
  }

  // 提按曲线:t(0..1)→ 宽度系数
  var PRESS = {
    zhong: function (t) {                   // 中锋:起笔按下、行笔略提、收笔回锋收细
      if (t < 0.08) return 0.45 + 0.55 * Math.sin(t / 0.08 * Math.PI / 2);
      if (t > 0.82) return 0.15 + 0.6 * Math.cos((t - 0.82) / 0.18 * Math.PI / 2);
      return 0.75;
    },
    leaf: function (t) { return 0.08 + 0.92 * Math.pow(Math.sin(t * Math.PI), 1.4); },   // 兰叶:细-粗-细
    dot: function (t) { return 0.55 + 0.45 * Math.sin(t * Math.PI); }                     // 点
  };

  function f1(n) { return Math.round(n * 10) / 10; }

  function create(opts) {
    var r = rng(opts.seed || 1), prefix = opts.id || 'b', n = 0;
    var art = [], defs = [], css = [];

    function stroke(pts, o) {
      var k = n++, width = o.width || 12, press = PRESS[o.press || 'zhong'];
      var load0 = o.load == null ? 0.95 : o.load, wet = o.wet == null ? 0.5 : o.wet;
      var start = o.start || 0, dur = o.dur || 1, tone = o.tone || 'nong';
      var P = resample(pts, 2), L = P[P.length - 1].s;
      var wob = noise1d(r, 6), edgeL = noise1d(r, 18), edgeR = noise1d(r, 18), loadN = noise1d(r, 5);
      var load = function (t) { return load0 - (load0 - 0.12) * Math.pow(t, 1.3) + 0.06 * loadN(t); };
      var W = function (t) { return width * press(t) * (1 + 0.08 * wob(t)); };
      var DRY = 0.42;                           // 含墨量低于它的地方改用笔毛画(飞白)
      var tDry = 1;
      for (var q = 0; q < P.length; q++) { if (load(P[q].s / L) < DRY) { tDry = P[q].s / L; break; } }

      function normal(i) {
        var a = P[Math.max(0, i - 1)], b = P[Math.min(P.length - 1, i + 1)];
        var dx = b.x - a.x, dy = b.y - a.y, len = Math.hypot(dx, dy) || 1;
        return [-dy / len, dx / len];
      }
      // 湿的部分:从起点画到 tDry 稍后一点,两边按宽度偏出去,边缘加一点抖
      var left = [], right = [], tWetEnd = Math.min(1, tDry + 0.04);
      for (var i = 0; i < P.length; i++) {
        var t = P[i].s / L;
        if (t > tWetEnd) break;
        var nrm = normal(i), w = W(t) / 2;
        left.push([P[i].x + nrm[0] * (w + 0.7 * edgeL(t)), P[i].y + nrm[1] * (w + 0.7 * edgeL(t))]);
        right.push([P[i].x - nrm[0] * (w + 0.7 * edgeR(t)), P[i].y - nrm[1] * (w + 0.7 * edgeR(t))]);
      }
      var body = '';
      if (left.length > 1) {
        var ring = left.concat(right.reverse());
        body = 'M' + ring.map(function (p) { return f1(p[0]) + ' ' + f1(p[1]); }).join('L') + 'Z';
      }
      // 干的部分:一束笔毛,每根在含墨不够的地方断开
      var N = Math.max(6, Math.min(24, Math.round(width / 2))), bristles = '';
      if (tDry < 1) {
        for (var b = 0; b < N; b++) {
          var u = (b + 0.5) / N - 0.5 + (r() - 0.5) * 0.04, bias = (r() - 0.5) * 0.25, bn = noise1d(r, 12);
          var seg = [], on = false;
          for (var j = 0; j < P.length; j++) {
            var tj = P[j].s / L;
            if (tj < tDry - 0.06) continue;
            var inkHere = load(tj) + bias + 0.22 * bn(tj) + 0.3 * (1 - Math.abs(u) * 2) * 0.5;
            var nm = normal(j), wj = W(tj);
            var x = P[j].x + nm[0] * u * wj, y = P[j].y + nm[1] * u * wj;
            if (inkHere > 0.33) { seg.push((on ? 'L' : 'M') + f1(x) + ' ' + f1(y)); on = true; }
            else on = false;
          }
          bristles += seg.join('');
        }
      }
      var center = 'M' + P.map(function (p) { return f1(p.x) + ' ' + f1(p.y); }).join('L');
      var mid = prefix + '-m' + k, wMax = width + 10;
      defs.push('<mask id="' + mid + '" maskUnits="userSpaceOnUse"><path class="bp-reveal" d="' + center +
        '" pathLength="1" stroke="#fff" stroke-width="' + wMax + '" fill="none" stroke-linecap="round"' +
        ' style="animation-delay:' + start + 's;animation-duration:' + dur + 's"/></mask>');
      var bw = f1(Math.max(0.8, width / N * 1.3));
```

```js
// 小样 2:把画在宣纸上的水墨位图抠成透明层(一次性,验证完就丢)。
//
// 做法(和 GIMP 的「颜色转透明」同一个思路):假设墨最深能到纯黑,每个像素取
// 「把纸色变成它,最少要盖多少墨」当透明度 a = max_c (P_c − C_c) / P_c,
// 再按「叠在纸上」反推墨的颜色 I_c = (C_c − (1 − a)·P_c) / a。
// 叠回同一张纸上,结果和原图逐像素相同;叠到别的底上,颜色也说得通。
// 纸上细小的颗粒(a < floor)直接清零,不然换个底色会带出一层灰。
//
// 用法:node layers.mjs extract <输入图> <输出.png> [--floor=0.04] [--paper=r,g,b]
//       node layers.mjs --selftest
// 本机 python 没有 PIL,所以像素处理放在 playwright 打开的浏览器里做。
import { createRequire } from 'node:module';
import { readFileSync, writeFileSync, mkdtempSync } from 'node:fs';
import { join } from 'node:path';
import { tmpdir } from 'node:os';

const SKY = process.env.SKY || process.cwd();
const require = createRequire(join(SKY, 'package.json'));
const { chromium } = require('playwright');
const { PNG } = require('pngjs');

// 在浏览器里跑的部分:估纸色、算透明度和墨色
function extractInPage({ dataUrl, floor, paper }) {
  return new Promise((resolve) => {
    const img = new Image();
    img.onload = () => {
      const c = document.createElement('canvas'); c.width = img.width; c.height = img.height;
      const g = c.getContext('2d'); g.drawImage(img, 0, 0);
      const d = g.getImageData(0, 0, c.width, c.height), px = d.data;
      // 纸色:亮度直方图里像素最多的那一档(水墨画留白多,这一档就是纸),
      // 取这一档 ±6 以内像素的各通道中位数。不用「最亮的一批」:那批落在纸纹的亮端,
      // 估出来偏亮,纸纹就被当成淡墨留了下来(2026-10-10 自测实测:白底上 4.4% 灰点)。
      const hist = new Array(256).fill(0), L = new Uint8Array(px.length / 4);
      for (let i = 0, k = 0; i < px.length; i += 4, k++) { L[k] = Math.round(0.2126 * px[i] + 0.7152 * px[i + 1] + 0.0722 * px[i + 2]); hist[L[k]]++; }
      let mode = 0; for (let v = 1; v < 256; v++) if (hist[v] > hist[mode]) mode = v;
      const top = []; for (let k = 0; k < L.length; k++) if (Math.abs(L[k] - mode) <= 6) top.push([0, k * 4]);
      const med = (ch) => { const v = top.map(([, i]) => px[i + ch]).sort((a, b) => a - b); return v[v.length >> 1]; };
      // 纸那一档像素太少、或者太暗,说明画面大半是墨(满幅泼墨),众数不是纸 —— 估出来的值不可信
      const share = top.length / L.length, suspicious = share < 0.15 || mode < 150;
      const P = paper || [med(0), med(1), med(2)];
      for (let i = 0; i < px.length; i += 4) {
        let a = 0;
        for (let ch = 0; ch < 3; ch++) a = Math.max(a, (P[ch] - px[i + ch]) / P[ch]);
        if (a < floor) { px[i + 3] = 0; continue; }
        a = Math.min(1, a);
        for (let ch = 0; ch < 3; ch++) px[i + ch] = Math.max(0, Math.min(255, Math.round((px[i + ch] - (1 - a) * P[ch]) / a)));
        px[i + 3] = Math.round(a * 255);
      }
      g.putImageData(d, 0, 0);
      resolve({ paper: P, share, mode, suspicious: !paper && suspicious, png: c.toDataURL('image/png') });
    };
    img.src = dataUrl;
  });
}

export async function extract(input, output, floor = 0.04, paper = null) {
  const browser = await chromium.launch();
  try {
    const page = await browser.newPage();
    const mime = input.endsWith('.png') ? 'image/png' : input.endsWith('.webp') ? 'image/webp' : 'image/jpeg';
    const dataUrl = `data:${mime};base64,` + readFileSync(input).toString('base64');
    const r = await page.evaluate(extractInPage, { dataUrl, floor, paper });
    if (r.suspicious) console.warn(`警告:像素最多的那一档亮度 ${r.mode}、占 ${(r.share * 100).toFixed(1)}%,看起来不是纸;估出的纸色 ${r.paper} 可能不对,画面大半是墨时请用 --paper=r,g,b 指定`);
    writeFileSync(output, Buffer.from(r.png.split(',')[1], 'base64'));
    return r;
  } finally { await browser.close(); }
}

// 把透明层叠到纯色底上(普通混合),返回 RGBA 缓冲
function over(layer, bg) {
  const out = Buffer.alloc(layer.data.length);
  for (let i = 0; i < layer.data.length; i += 4) {
    const a = layer.data[i + 3] / 255;
    for (let ch = 0; ch < 3; ch++) out[i + ch] = Math.round(layer.data[i + ch] * a + bg[ch] * (1 - a));
    out[i + 3] = 255;
  }
  return out;
}

async function selftest() {
  const dir = mkdtempSync(join(tmpdir(), 'layers-'));
  const W = 240, H = 160, paper = [242, 234, 216];
  const src = new PNG({ width: W, height: H }), inkMask = new Uint8Array(W * H);
  let seed = 7; const r = () => ((seed = (seed * 1103515245 + 12345) >>> 0) / 4294967296);
  for (let y = 0; y < H; y++) for (let x = 0; x < W; x++) {
    const i = (y * W + x) * 4, grain = (r() - 0.5) * 10;         // 纸上 ±5 的颗粒(去掉清零那步时,白底上会留下灰点)
    let c = paper.map((v) => v + grain);
    if (Math.hypot(x - 70, y - 80) < 34) { c = c.map((v) => v * 0.15); inkMask[y * W + x] = 1; }            // 一团焦墨
    else if (x > 130 && x < 220 && y > 40 && y < 120) { c = [c[0] * 0.55, c[1] * 0.7, c[2] * 0.9]; inkMask[y * W + x] = 1; } // 一片花青
    for (let ch = 0; ch < 3; ch++) src.data[i + ch] = Math.max(0, Math.min(255, Math.round(c[ch])));
    src.data[i + 3] = 255;
  }
  const inPath = join(dir, 'in.png'), outPath = join(dir, 'out.png');
  writeFileSync(inPath, PNG.sync.write(src));
  const est = (await extract(inPath, outPath, 0.04)).paper;
  const layer = PNG.sync.read(readFileSync(outPath));
  let fail = 0;
  const check = (name, ok, detail) => { console.log((ok ? 'PASS ' : 'FAIL ') + name + ' — ' + detail); if (!ok) fail = 1; };

  check('估出的纸色接近真纸色', est.every((v, k) => Math.abs(v - paper[k]) <= 3), `估 ${est} / 真 ${paper}`);
  // 叠回原纸色:有墨的地方逐像素接近原图
  const back = over(layer, est); let sum = 0, cnt = 0;
  for (let p = 0; p < W * H; p++) if (inkMask[p]) { for (let ch = 0; ch < 3; ch++) sum += Math.abs(back[p * 4 + ch] - src.data[p * 4 + ch]); cnt += 3; }
  check('叠回原纸,墨的部分和原图一样', sum / cnt < 1.5, `平均差 ${(sum / cnt).toFixed(2)}`);
  // 叠到磁青纸上:原本是纸的地方不能留下灰色残留
  const dark = [20, 33, 58], onDark = over(layer, dark); let residue = 0, paperPx = 0;
  for (let p = 0; p < W * H; p++) if (!inkMask[p]) {
    paperPx++;
    const diff = Math.max(...[0, 1, 2].map((ch) => Math.abs(onDark[p * 4 + ch] - dark[ch])));
    if (diff > 8) residue++;
  }
  check('换到深底,纸的部分没有残留', residue / paperPx < 0.005, `残留 ${(100 * residue / paperPx).toFixed(2)}%`);

  const white = [255, 255, 255], onWhite = over(layer, white); let grey = 0;
  for (let p = 0; p < W * H; p++) if (!inkMask[p]) {
    const diff = Math.max(...[0, 1, 2].map((ch) => Math.abs(onWhite[p * 4 + ch] - white[ch])));
    if (diff > 4) grey++;                 // 纸纹 ±5 不清零时,灰点偏差约 5–6,超过 4
```

```js
    b.stroke([[x, y], [mx, my], [x + Math.cos(ang) * len, y + Math.sin(ang) * len]],
      { width: 8 + r() * 10, press: 'leaf', load: 0.5 + r() * 0.45, tone: ['nong', 'zhong', 'dan'][i % 3], start: i * dur / n, dur: 0.35 });
  }

  const f = path.join(__dirname, 'perf-' + n + '.html'); fs.writeFileSync(f, html);
```

```python
# 生成两种减负版本的画笔,和原版对比性能:
#   brush-proto.nofilter.js  去掉整组打毛滤镜,边缘抖动全放进几何(0.7 → 1.2)
#   brush-proto.unmask.js    每笔画完后用一个 1ms 的 CSS 动画把遮罩换成 none
import pathlib
src = pathlib.Path('brush-proto.js').read_text()

s = src
for a, b in [
    ("'<g class=\"bp-ink\" filter=\"url(#bp-rough)\">'", "'<g class=\"bp-ink\">'"),
    ("0.7 * edgeL(t)", "1.2 * edgeL(t)"),
    ("0.7 * edgeR(t)", "1.2 * edgeR(t)"),
]:
    assert a in s, a
    s = s.replace(a, b)
pathlib.Path('brush-proto.nofilter.js').write_text(s)

s = src
for a, b in [
    ("art.push('<g class=\"bp-stroke bp-' + tone + '\" mask=\"url(#' + mid + ')\">' +",
     "art.push('<g class=\"bp-stroke bp-' + tone + '\" mask=\"url(#' + mid + ')\" style=\"animation-delay:' + f1(start + dur) + 's\">' +"),
    ("'.bp-reveal{stroke-dasharray",
     "'.bp-stroke{animation:bp-unmask 1ms step-end forwards}@keyframes bp-unmask{to{mask:none}}' +\n    '.bp-reveal{stroke-dasharray"),
]:
    assert a in s, a
    s = s.replace(a, b)
pathlib.Path('brush-proto.unmask.js').write_text(s)
print('写出 brush-proto.nofilter.js brush-proto.unmask.js')
```
